# pipeline/loader.py
from dataclasses import dataclass
import pandas as pd

DATA_PATH = "260330_수불_더미데이터.xlsx"
REF_PATH = "260330_기준정보.xlsx"


@dataclass
class LoadedData:
    receipt: pd.DataFrame
    usage: pd.DataFrame
    exp_receipt: pd.DataFrame
    exp_usage: pd.DataFrame
    opening: pd.DataFrame
    ref: pd.DataFrame


def load_workbooks(
    data_path: str = DATA_PATH,
    ref_path: str = REF_PATH,
) -> LoadedData:
    wb = pd.ExcelFile(data_path)
    return LoadedData(
        receipt=pd.read_excel(wb, sheet_name="입고"),
        usage=pd.read_excel(wb, sheet_name="사용"),
        exp_receipt=pd.read_excel(wb, sheet_name="예상입고량"),
        exp_usage=pd.read_excel(wb, sheet_name="예상사용량"),
        opening=pd.read_excel(wb, sheet_name="기초재고"),
        ref=pd.read_excel(ref_path, sheet_name="기준정보"),
    )
```

# pipeline/transform.py
from datetime import date as date_type
import pandas as pd

SITE_MAP = {
    "광양": "광양소",
    "포항": "포항소",
    "광양소": "광양소",
    "포항소": "포항소",
}


def standardize_movement_df(
    df: pd.DataFrame,
    qty_col: str,
    date_col: str,
) -> pd.DataFrame:
    """입고/사용 실적 DataFrame 표준화.

    반환 컬럼: 사소구분, 구매item, date, qty
    입고의 경우 등급·공급사 컬럼이 있으면 유지.
    """
    out = df.copy()
    out["사소구분"] = out["사소구분"].map(SITE_MAP)
    out["date"] = pd.to_datetime(out[date_col]).dt.date
    out["qty"] = out[qty_col].fillna(0).astype(float)

    keep = ["사소구분", "구매item", "date", "qty"]
    for extra in ["등급", "공급사"]:
        if extra in out.columns:
            keep.append(extra)
    return out[keep].reset_index(drop=True)


def standardize_expected_df(df: pd.DataFrame) -> pd.DataFrame:
    """예상 데이터 Wide → Long 변환.

    반환 컬럼: 사소구분, 구매item, date, qty
    """
    out = df.copy()
    out["사소구분"] = out["사소구분"].map(SITE_MAP)

    id_cols = ["사소구분", "ITEM"]
    date_cols = [c for c in out.columns if c not in id_cols]

    melted = out.melt(
        id_vars=id_cols,
        value_vars=date_cols,
        var_name="date",
        value_name="qty",
    )
    melted["date"] = pd.to_datetime(melted["date"]).dt.date
    melted["qty"] = melted["qty"].fillna(0).astype(float)
    melted = melted.rename(columns={"ITEM": "구매item"})
    return melted[["사소구분", "구매item", "date", "qty"]].reset_index(drop=True)
```

# pipeline/metrics.py
import pandas as pd
from pipeline.loader import LoadedData
from pipeline.transform import standardize_movement_df, standardize_expected_df


def bucket_metrics(data: LoadedData) -> dict[str, pd.DataFrame]:
    """LoadedData → 사소×ITEM×날짜 집계 결과."""
    receipt = standardize_movement_df(data.receipt, qty_col="입하량(net)", date_col="입하일시")
    usage = standardize_movement_df(data.usage, qty_col="입하량(net)", date_col="입하일시")
    exp_receipt = standardize_expected_df(data.exp_receipt)
    exp_usage = standardize_expected_df(data.exp_usage)

    # opening의 컬럼명 통일 (구매ITEM → 구매item)
    opening = data.opening.copy()
    opening = opening.rename(columns={"구매ITEM": "구매item"})
    opening["사소구분"] = opening["사소구분"].map(
        {"광양": "광양소", "포항": "포항소", "광양소": "광양소", "포항소": "포항소"}
    )

    daily = daily_recv_use_inv(receipt, usage, exp_receipt, exp_usage, opening)
    return {"daily": daily}


def daily_recv_use_inv(
    receipt: pd.DataFrame,
    usage: pd.DataFrame,
    exp_receipt: pd.DataFrame,
    exp_usage: pd.DataFrame,
    opening: pd.DataFrame,
) -> pd.DataFrame:
    """일별 입고/사용/재고 계산.

    - receipt, usage: standardize_movement_df 결과 (컬럼: 사소구분, 구매item, date, qty)
    - exp_receipt, exp_usage: standardize_expected_df 결과 (동일 구조)
    - opening: 사소구분, 구매item, 재고량
    반환: 사소구분, 구매item, date, recv_qty, use_qty, inv, is_actual
    """
    # 실적 일별 집계
    recv_act = (
        receipt.groupby(["사소구분", "구매item", "date"])["qty"]
        .sum()
        .reset_index()
        .rename(columns={"qty": "recv_qty"})
    )
    use_act = (
        usage.groupby(["사소구분", "구매item", "date"])["qty"]
        .sum()
        .reset_index()
        .rename(columns={"qty": "use_qty"})
    )

    # 실적이 있는 (사소, item, date) 집합
    recv_act_keys = set(zip(recv_act["사소구분"], recv_act["구매item"], recv_act["date"]))
    use_act_keys = set(zip(use_act["사소구분"], use_act["구매item"], use_act["date"]))

    # 예상 데이터에서 실적이 있는 날 제거
    exp_recv_fil = exp_receipt[
        ~exp_receipt.apply(
            lambda r: (r["사소구분"], r["구매item"], r["date"]) in recv_act_keys, axis=1
        )
    ].rename(columns={"qty": "recv_qty"})

    exp_use_fil = exp_usage[
        ~exp_usage.apply(
            lambda r: (r["사소구분"], r["구매item"], r["date"]) in use_act_keys, axis=1
        )
    ].rename(columns={"qty": "use_qty"})

    # 입고 병합 (실적 + 예상)
    recv_act["is_actual"] = True
    exp_recv_fil = exp_recv_fil.copy()
    exp_recv_fil["is_actual"] = False
    recv_all = pd.concat(
        [recv_act[["사소구분", "구매item", "date", "recv_qty", "is_actual"]],
         exp_recv_fil[["사소구분", "구매item", "date", "recv_qty", "is_actual"]]],
        ignore_index=True,
    )

    # 사용 병합 (실적 + 예상)
    use_act["is_actual"] = True
    exp_use_fil = exp_use_fil.copy()
    exp_use_fil["is_actual"] = False
    use_all = pd.concat(
        [use_act[["사소구분", "구매item", "date", "use_qty", "is_actual"]],
         exp_use_fil[["사소구분", "구매item", "date", "use_qty", "is_actual"]]],
        ignore_index=True,
    )

    # 입고+사용 outer join
    daily = pd.merge(
        recv_all, use_all,
        on=["사소구분", "구매item", "date"],
        how="outer",
        suffixes=("_r", "_u"),
    )
    daily["recv_qty"] = daily["recv_qty"].fillna(0)
    daily["use_qty"] = daily["use_qty"].fillna(0)
    daily["is_actual"] = daily["is_actual_r"].fillna(False) | daily["is_actual_u"].fillna(False)
    daily = daily.drop(columns=["is_actual_r", "is_actual_u"])
    daily = daily.sort_values(["사소구분", "구매item", "date"]).reset_index(drop=True)

    # 기초재고 lookup
    opening_map = (
        opening.set_index(["사소구분", "구매item"])["재고량"].to_dict()
    )

    # 사소×ITEM 그룹별 누적 재고 계산
    parts = []
    for (site, item), grp in daily.groupby(["사소구분", "구매item"]):
        grp = grp.sort_values("date").copy()
        init = opening_map.get((site, item), 0.0)
        grp["inv"] = init + (grp["recv_qty"].cumsum() - grp["use_qty"].cumsum())
        parts.append(grp)

    result = pd.concat(parts, ignore_index=True)
    return result[["사소구분", "구매item", "date", "recv_qty", "use_qty", "inv", "is_actual"]]
```

# pipeline/supplier.py
import pandas as pd
from pipeline.transform import standardize_movement_df, SITE_MAP


def build_supplier_summary(
    receipt: pd.DataFrame,
    ref: pd.DataFrame,
) -> pd.DataFrame:
    """입고 실적과 기준정보를 매칭하여 공급사별 입고량 집계 반환.

    반환 컬럼: 공급사명, 구분, ITEM, 사소구분, recv_qty
    미매칭 공급사는 공급사명='기타', 구분=None, ITEM=None 으로 처리.
    """
    std = standardize_movement_df(receipt, qty_col="입하량(net)", date_col="입하일시")

    # 공급사명으로 LEFT JOIN
    merged = std.merge(
        ref.rename(columns={"공급사명": "공급사"}),
        on="공급사",
        how="left",
    )

    # 미매칭 처리
    merged["공급사명"] = merged["공급사"].where(merged["구분"].notna(), "기타")
    merged["구분"] = merged["구분"].fillna("미분류")
    merged["ITEM"] = merged["ITEM"].where(merged["ITEM"].notna(), merged["구매item"])

    summary = (
        merged.groupby(["공급사명", "구분", "ITEM", "사소구분"])["qty"]
        .sum()
        .reset_index()
        .rename(columns={"qty": "recv_qty"})
    )
    return summary
```

def make_monthly_chart(daily: pd.DataFrame) -> go.Figure:
    """월간 추이 복합 차트.

    - 선그래프: 입고량(실적=실선, 예상=점선+연한색)
    - 선그래프: 사용량(실적=실선, 예상=점선+연한색)
    - 막대그래프: 재고량(실적/예상 색상 구분)
    """

        return df.groupby("date")[col].sum().reset_index()

    fig = go.Figure()

    # 재고 막대 (뒤에 렌더)
    inv_act = day_sum(actual, "inv")
    inv_fct = day_sum(forecast, "inv")

        x=inv_act["date"], y=inv_act["inv"],
        name="재고(실적)", marker_color=ACTUAL_COLOR_INV, opacity=0.6,
    ))
    fig.add_trace(go.Bar(
        x=inv_fct["date"], y=inv_fct["inv"],
        name="재고(예상)", marker_color=FORECAST_INV, opacity=0.4,
    ))

    # 입고 선
    recv_act = day_sum(actual, "recv_qty")
    recv_fct = day_sum(forecast, "recv_qty")

    fig.add_trace(go.Scatter(
        x=recv_act["date"], y=recv_act["recv_qty"],
        name="입고(실적)", mode="lines+markers",
        line=dict(color=ACTUAL_COLOR_RECV, width=2),
    ))
    fig.add_trace(go.Scatter(
        x=recv_fct["date"], y=recv_fct["recv_qty"],
        name="입고(예상)", mode="lines+markers",
        line=dict(color=ACTUAL_COLOR_RECV, width=2, dash="dot"),
        opacity=FORECAST_ALPHA,
    ))

    # 사용 선
    use_act = day_sum(actual, "use_qty")
    use_fct = day_sum(forecast, "use_qty")

    fig.add_trace(go.Scatter(
        x=use_act["date"], y=use_act["use_qty"],
        name="사용(실적)", mode="lines+markers",
        line=dict(color=ACTUAL_COLOR_USE, width=2),
    ))
    fig.add_trace(go.Scatter(
        x=use_fct["date"], y=use_fct["use_qty"],
        name="사용(예상)", mode="lines+markers",

        opacity=FORECAST_ALPHA,
    ))

    # 실적↔예상 경계선
    if not actual.empty and not forecast.empty:
        boundary = actual["date"].max()
        fig.add_vline(
            x=str(boundary),
            line_dash="dash", line_color="gray",
            annotation_text="실적↔예상", annotation_position="top right",
        )

    fig.update_layout(
        barmode="overlay",
        xaxis_title="날짜",
        yaxis_title="수량",
        legend=dict(orientation="h", yanchor="bottom", y=1.02, xanchor="right", x=1),
        hovermode="x unified",
    )
    return fig


def make_item_chart(daily: pd.DataFrame) -> go.Figure:

# report/exporter.py
import io
import pandas as pd
import xlsxwriter


def export_workbook_bytes(
    daily: pd.DataFrame,
    supplier: pd.DataFrame,
) -> bytes:
    """daily 집계와 supplier 집계를 Excel bytes로 반환.

    시트 구성:
      - 수불현황: 날짜별 입고→사용→재고 (포항소/광양소 컬럼 분리)
      - 공급사: 공급사×ITEM×사소 집계
    """
    buf = io.BytesIO()

    with pd.ExcelWriter(buf, engine="xlsxwriter") as writer:
        _write_subbul_sheet(writer, daily)
        _write_supplier_sheet(writer, supplier)

    return buf.getvalue()


def _write_subbul_sheet(writer: pd.ExcelWriter, daily: pd.DataFrame) -> None:
    sites = sorted(daily["사소구분"].unique())
    rows = []

    # 입고 섹션
    recv = daily.pivot_table(
        index="date", columns="사소구분", values="recv_qty", aggfunc="sum", fill_value=0
    ).reset_index()
    recv.insert(0, "항목", "입고")
    rows.append(recv)

    # 사용 섹션
    use = daily.pivot_table(
        index="date", columns="사소구분", values="use_qty", aggfunc="sum", fill_value=0
    ).reset_index()
    use.insert(0, "항목", "사용")
    rows.append(use)

    # 재고 섹션
    inv = daily.pivot_table(
        index="date", columns="사소구분", values="inv", aggfunc="sum", fill_value=0
    ).reset_index()
    inv.insert(0, "항목", "재고")
    rows.append(inv)

    out = pd.concat(rows, ignore_index=True)
    out.to_excel(writer, sheet_name="수불현황", index=False)

    # 실적/예상 행 색상
    wb = writer.book
    ws = writer.sheets["수불현황"]
    actual_fmt = wb.add_format({"bg_color": "#FFFFFF"})
    forecast_fmt = wb.add_format({"bg_color": "#FFF3CD"})

    # is_actual 정보로 행별 서식 적용
    is_actual_map = (
        daily.groupby("date")["is_actual"].all().to_dict()
    )
    for row_idx, date_val in enumerate(out["date"], start=1):
        fmt = actual_fmt if is_actual_map.get(date_val, True) else forecast_fmt
        ws.set_row(row_idx, None, fmt)


def _write_supplier_sheet(writer: pd.ExcelWriter, supplier: pd.DataFrame) -> None:
    pivot = supplier.pivot_table(
        index=["공급사명", "구분", "ITEM"],
        columns="사소구분",
        values="recv_qty",
        aggfunc="sum",
        fill_value=0,
    ).reset_index()
    pivot["합계"] = pivot.select_dtypes("number").sum(axis=1)
    pivot.to_excel(writer, sheet_name="공급사", index=False)
```

        mask &= receipt_raw["등급"].isin(f_grade)

    filtered = receipt_raw[mask]

    st.dataframe(filtered.reset_index(drop=True), use_container_width=True)

    if not filtered.empty:
        # 공급사별 입고 차트 (filtered 기준 직접 집계)
        sup_agg = filtered.groupby("공급사")["입하량(net)"].sum().sort_values().reset_index()
        sup_agg.columns = ["공급사명", "recv_qty"]
        st.plotly_chart(make_supplier_bar(sup_agg), use_container_width=True)

    # ── 다운로드 ──────────────────────────────────────────────
    st.divider()
    excel_bytes = export_workbook_bytes(daily_f, supplier_f)
    st.download_button(
        label="Excel 다운로드",
        data=excel_bytes,
        file_name="수불현황.xlsx",
        mime="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
    )
```

> **주의:** `make_supplier_bar`가 `공급사명` 컬럼을 요구하므로 입고 상세 탭에서는 `공급사` 컬럼을 `공급사명`으로 rename해서 전달해야 합니다. 위 코드에서 `sup_agg` 변수가 이를 처리합니다.
